package com.example.engine

import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.charset.StandardCharsets

object BinaryXmlDecoder {
    private const val CHUNK_AXML_FILE = 0x00080003
    private const val CHUNK_STRING_POOL = 0x001C0001
    private const val CHUNK_RESOURCE_IDS = 0x00080180
    private const val CHUNK_START_NAMESPACE = 0x00100100
    private const val CHUNK_END_NAMESPACE = 0x00100101
    private const val CHUNK_START_TAG = 0x00100102
    private const val CHUNK_END_TAG = 0x00100103
    private const val CHUNK_TEXT = 0x00100104

    fun isBinaryXml(bytes: ByteArray): Boolean {
        if (bytes.size < 8) return false
        val magic = ByteBuffer.wrap(bytes, 0, 4).order(ByteOrder.LITTLE_ENDIAN).int
        return magic == CHUNK_AXML_FILE || (magic and 0xFFFF) == 0x0003
    }

    fun decode(bytes: ByteArray): String {
        if (bytes.isEmpty()) return ""
        if (!isBinaryXml(bytes)) return String(bytes, StandardCharsets.UTF_8)
        return try {
            val buf = ByteBuffer.wrap(bytes).order(ByteOrder.LITTLE_ENDIAN)
            val fileType = buf.int
            if ((fileType and 0xFFFF) != 0x0003) return String(bytes, StandardCharsets.UTF_8)
            val fileSize = buf.int.coerceAtMost(bytes.size)
            var strings: List<String> = emptyList()
            val namespaces = mutableMapOf<String, String>()
            val pendingNs = mutableListOf<Pair<String, String>>()
            val out = StringBuilder("<?xml version=\"1.0\" encoding=\"utf-8\"?>\n")
            var indent = 0
            while (buf.position() + 8 <= fileSize) {
                val chunkStart = buf.position()
                val chunkType = buf.int
                val chunkSize = buf.int
                if (chunkSize < 8 || chunkStart + chunkSize > bytes.size) break
                when (chunkType) {
                    CHUNK_STRING_POOL, 0x0001 -> strings = parseStringPool(bytes, chunkStart, chunkSize)
                    CHUNK_RESOURCE_IDS -> {}
                    CHUNK_START_NAMESPACE -> if (chunkSize >= 24) {
                        buf.int; buf.int
                        val prefix = strings.getOrNull(buf.int) ?: "ns"
                        val uri = strings.getOrNull(buf.int) ?: ""
                        if (uri.isNotEmpty()) { namespaces[uri] = prefix; pendingNs.add(prefix to uri) }
                    }
                    CHUNK_END_NAMESPACE -> {}
                    CHUNK_START_TAG -> if (chunkSize >= 36) {
                        buf.int; buf.int
                        val nsIdx = buf.int; val nameIdx = buf.int
                        val attrStart = buf.short.toInt() and 0xFFFF
                        val attrSize = buf.short.toInt() and 0xFFFF
                        val attrCount = buf.short.toInt() and 0xFFFF
                        buf.short; buf.short; buf.short
                        val tagName = strings.getOrNull(nameIdx) ?: "tag"
                        val tagPrefix = if (nsIdx >= 0) strings.getOrNull(nsIdx)?.let { namespaces[it] } else null
                        val fullTag = if (!tagPrefix.isNullOrEmpty()) "$tagPrefix:$tagName" else tagName
                        repeat(indent.coerceAtMost(20)) { out.append("  ") }
                        out.append("<").append(fullTag)
                        for ((p, u) in pendingNs) out.append(" xmlns:$p=\"${escapeXml(u)}\"")
                        pendingNs.clear()
                        val attrOffset = chunkStart + 16 + attrStart
                        val stride = if (attrSize >= 20) attrSize else 20
                        for (i in 0 until attrCount) {
                            val pos = attrOffset + i * stride
                            if (pos + 20 > chunkStart + chunkSize) break
                            buf.position(pos)
                            val aNs = buf.int; val aNameIdx = buf.int; val rawIdx = buf.int
                            buf.short; buf.get()
                            val vType = buf.get().toInt() and 0xFF
                            val vData = buf.int
                            val aName = strings.getOrNull(aNameIdx) ?: "attr_$i"
                            val aPrefix = if (aNs >= 0) strings.getOrNull(aNs)?.let { namespaces[it] } else null
                            val fullAttr = if (!aPrefix.isNullOrEmpty()) "$aPrefix:$aName" else aName
                            val vStr = if (rawIdx in strings.indices) strings[rawIdx] else when (vType) {
                                0x03 -> strings.getOrNull(vData) ?: ""
                                0x01 -> "@0x${Integer.toHexString(vData)}"
                                0x02 -> "?0x${Integer.toHexString(vData)}"
                                0x12 -> if (vData != 0) "true" else "false"
                                0x10 -> vData.toString()
                                else -> if (aName.contains("SdkVersion", true)) vData.toString() else "0x${Integer.toHexString(vData)}"
                            }
                            if (attrCount > 2) { out.append("\n"); repeat((indent + 1).coerceAtMost(20)) { out.append("  ") } } else out.append(" ")
                            out.append("$fullAttr=\"${escapeXml(vStr)}\"")
                        }
                        out.append(">\n"); indent++
                    }
                    CHUNK_END_TAG -> if (chunkSize >= 24) {
                        buf.int; buf.int
                        val nsIdx = buf.int; val nameIdx = buf.int
                        val tagName = strings.getOrNull(nameIdx) ?: "tag"
                        val tagPrefix = if (nsIdx >= 0) strings.getOrNull(nsIdx)?.let { namespaces[it] } else null
                        val fullTag = if (!tagPrefix.isNullOrEmpty()) "$tagPrefix:$tagName" else tagName
                        indent = (indent - 1).coerceAtLeast(0)
                        repeat(indent.coerceAtMost(20)) { out.append("  ") }
                        out.append("</$fullTag>\n")
                    }
                    CHUNK_TEXT -> if (chunkSize >= 28) {
                        buf.int; buf.int
                        val text = strings.getOrNull(buf.int)
                        if (!text.isNullOrBlank()) {
                            repeat(indent.coerceAtMost(20)) { out.append("  ") }
                            out.append(escapeXml(text.trim())).append("\n")
                        }
                    }
                }
                buf.position(chunkStart + chunkSize)
            }
            out.toString()
        } catch (e: Exception) { "<!-- Failed to decode binary XML: ${e.message} -->\n" }
    }

    private fun parseStringPool(bytes: ByteArray, chunkStart: Int, chunkSize: Int): List<String> {
        val buf = ByteBuffer.wrap(bytes, chunkStart, chunkSize).order(ByteOrder.LITTLE_ENDIAN)
        buf.int; buf.int
        val count = buf.int; buf.int
        val flags = buf.int; val stringsStart = buf.int; buf.int
        val isUtf8 = (flags and (1 shl 8)) != 0
        val offsets = IntArray(count) { buf.int }
        val base = chunkStart + stringsStart
        return List(count) { i ->
            var pos = base + offsets[i]
            if (pos < 0 || pos >= bytes.size) "" else if (isUtf8) {
                pos += if ((bytes[pos].toInt() and 0x80) != 0) 2 else 1
                if (pos >= bytes.size) "" else {
                    var u8 = bytes[pos].toInt() and 0xFF
                    if ((u8 and 0x80) != 0 && pos + 1 < bytes.size) { u8 = ((u8 and 0x7F) shl 8) or (bytes[pos + 1].toInt() and 0xFF); pos += 2 } else pos += 1
                    String(bytes, pos, u8.coerceIn(0, (bytes.size - pos).coerceAtLeast(0)), StandardCharsets.UTF_8)
                }
            } else if (pos + 2 > bytes.size) "" else {
                var len = ((bytes[pos + 1].toInt() and 0xFF) shl 8) or (bytes[pos].toInt() and 0xFF)
                if ((len and 0x8000) != 0 && pos + 4 <= bytes.size) {
                    val low = ((bytes[pos + 3].toInt() and 0xFF) shl 8) or (bytes[pos + 2].toInt() and 0xFF)
                    len = ((len and 0x7FFF) shl 16) or low; pos += 4
                } else pos += 2
                String(bytes, pos, (len * 2).coerceIn(0, (bytes.size - pos).coerceAtLeast(0)), StandardCharsets.UTF_16LE)
            }
        }
    }

    private fun escapeXml(input: String) = input.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;").replace("\"", "&quot;")
}

        val folderSuffix = if (!subFolder.isNullOrBlank()) "/$subFolder" else ""
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            val values = ContentValues().apply {
                put(MediaStore.MediaColumns.DISPLAY_NAME, fileName)
                put(MediaStore.MediaColumns.MIME_TYPE, mimeType)
                put(MediaStore.MediaColumns.RELATIVE_PATH, "${Environment.DIRECTORY_DOWNLOADS}/CodeLens$folderSuffix")
                put(MediaStore.MediaColumns.IS_PENDING, 1)
            }
            val resolver = context.contentResolver
            val uri = resolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values) ?: error("Failed to create download")
            resolver.openOutputStream(uri)?.use { it.write(bytes) } ?: error("Failed to write file")
            values.clear(); values.put(MediaStore.MediaColumns.IS_PENDING, 0); resolver.update(uri, values, null, null)

        } else {
            val dir = File(context.getExternalFilesDir(Environment.DIRECTORY_DOWNLOADS), "CodeLens$folderSuffix").apply { mkdirs() }
            val outFile = File(dir, fileName)
            outFile.writeBytes(bytes)
            return outFile.absolutePath
        }
    }

object DexInspector {
    fun inspectMultipleDex(dexBytesList: List<Pair<String, ByteArray>>): DexAnalysisResult {
        val summaries = mutableListOf<DexFileSummary>()
        val packageToClasses = linkedMapOf<String, LinkedHashSet<String>>()
        val frameworks = linkedSetOf<String>()
        val urls = linkedSetOf<String>()
        var totalClasses = 0; var totalMethods = 0; var totalStrings = 0
        for ((fileName, bytes) in dexBytesList) {
            val parsed = inspectSingleDex(fileName, bytes) ?: continue
            summaries.add(parsed.summary)
            totalClasses += parsed.summary.classCount
            totalMethods += parsed.summary.methodCount
            totalStrings += parsed.summary.stringCount
            frameworks.addAll(parsed.frameworks)
            urls.addAll(parsed.urls)
            for (desc in parsed.classNames) {
                val fqcn = descriptorToClassName(desc)
                if (fqcn.isBlank()) continue
                val pkg = fqcn.substringBeforeLast('.', "(default)")
                val set = packageToClasses.getOrPut(pkg) { linkedSetOf() }
                if (set.size < 40) set.add(fqcn.substringAfterLast('.'))
            }
        }
        val sortedPackages = packageToClasses.entries.sortedWith(
            compareByDescending<Map.Entry<String, LinkedHashSet<String>>> {
                !it.key.startsWith("androidx.") && !it.key.startsWith("android.") && !it.key.startsWith("kotlin") && !it.key.startsWith("com.google.")
            }.thenByDescending { it.value.size }
        ).take(45).map { (pkg, classes) -> DexPackageSummary(pkg, classes.size, classes.toList().take(12)) }
        val outline = buildString {
            appendLine("// Extracted DEX Class & Package Outline")
            appendLine("// Total DEX files: ${summaries.size}, Classes: $totalClasses, Methods: $totalMethods\n")
            sortedPackages.take(25).forEach { pkg ->
                appendLine("package ${pkg.packageName};")
                pkg.sampleClasses.take(8).forEach { cls -> appendLine("    public class $cls { /* Dalvik bytecode */ }") }
                appendLine()
            }
        }
        return DexAnalysisResult(summaries, totalClasses, totalMethods, totalStrings, sortedPackages, frameworks, outline, urls.take(25))
    }

    fun formatDexFilePreview(fileName: String, bytes: ByteArray): String {
        val parsed = inspectSingleDex(fileName, bytes) ?: return "Invalid DEX file: $fileName"
        val s = parsed.summary
        return buildString {

            appendLine("// Size: ${s.fileSizeBytes} bytes | Classes: ${s.classCount} | Methods: ${s.methodCount} | Strings: ${s.stringCount}\n")
            parsed.classNames.take(250).forEach { desc ->
                val fqcn = descriptorToClassName(desc)
                if (fqcn.isNotEmpty()) appendLine("class $fqcn")
            }
        }
    }

    private data class SingleDexParsed(val summary: DexFileSummary, val classNames: List<String>, val frameworks: Set<String>, val urls: Set<String>)

    private fun inspectSingleDex(fileName: String, bytes: ByteArray): SingleDexParsed? {
        if (bytes.size < 112 || bytes[0] != 'd'.code.toByte() || bytes[1] != 'e'.code.toByte() || bytes[2] != 'x'.code.toByte() || bytes[3] != '\n'.code.toByte()) return null
        val version = String(bytes, 4, 3, Charsets.US_ASCII)
        val buf = ByteBuffer.wrap(bytes).order(ByteOrder.LITTLE_ENDIAN)
        val fileSize = buf.getInt(32); val stringIdsSize = buf.getInt(56).coerceAtLeast(0); val stringIdsOff = buf.getInt(60).coerceAtLeast(0)
        val typeIdsSize = buf.getInt(64).coerceAtLeast(0); val typeIdsOff = buf.getInt(68).coerceAtLeast(0)
        val fieldIdsSize = buf.getInt(80).coerceAtLeast(0); val methodIdsSize = buf.getInt(88).coerceAtLeast(0)
        val classDefsSize = buf.getInt(96).coerceAtLeast(0); val classDefsOff = buf.getInt(100).coerceAtLeast(0)
        val frameworks = linkedSetOf<String>(); val urls = linkedSetOf<String>(); val classNames = ArrayList<String>(minOf(classDefsSize, 1500))
        fun readStr(idx: Int): String? {
            if (idx !in 0 until stringIdsSize) return null
            val offPos = stringIdsOff + idx * 4; if (offPos + 4 > bytes.size) return null
            val dataOff = buf.getInt(offPos); if (dataOff !in 0 until bytes.size) return null
            var pos = dataOff
            while (pos < bytes.size) { val b = bytes[pos++].toInt() and 0xFF; if ((b and 0x80) == 0) break }
            val start = pos; val maxEnd = (start + 512).coerceAtMost(bytes.size)
            while (pos < maxEnd && bytes[pos].toInt() != 0) pos++
            return String(bytes, start, (pos - start).coerceAtLeast(0), Charsets.UTF_8)
        }
        for (i in 0 until minOf(classDefsSize, 2500)) {
            val cdPos = classDefsOff + i * 32; if (cdPos + 4 > bytes.size) break
            val typeIdx = buf.getInt(cdPos); if (typeIdx !in 0 until typeIdsSize) continue
            val tPos = typeIdsOff + typeIdx * 4; if (tPos + 4 > bytes.size) continue
            val desc = readStr(buf.getInt(tPos)) ?: continue
            classNames.add(desc); checkMarker(desc, frameworks)
        }
        val step = if (stringIdsSize > 18000) stringIdsSize / 18000 else 1
        var idx = 0
        while (idx < stringIdsSize) {
            val s = readStr(idx)
            if (s != null) {
                checkMarker(s, frameworks)
                if ((s.startsWith("https://") || s.startsWith("http://")) && s.length in 10..140 && !s.contains("schemas.android.com") && urls.size < 30) urls.add(s)
            }
            idx += step
        }
        return SingleDexParsed(DexFileSummary(fileName, version, fileSize, stringIdsSize, typeIdsSize, fieldIdsSize, methodIdsSize, classDefsSize), classNames, frameworks, urls)
    }

    private fun checkMarker(str: String, frameworks: MutableSet<String>) {
        when {
            str.contains("Lkotlin/Metadata;") || str.startsWith("Lkotlin/") || str.startsWith("Lkotlinx/") -> frameworks.add("Kotlin")
            str.startsWith("Landroidx/compose/") -> frameworks.add("Jetpack Compose")
            str.startsWith("Lio/flutter/") -> frameworks.add("Flutter")
            str.startsWith("Lcom/facebook/react/") -> frameworks.add("React Native")
            str.startsWith("Lorg/apache/cordova/") || str.startsWith("Lcom/getcapacitor/") -> frameworks.add("Cordova / Capacitor")
            str.startsWith("Landroid/webkit/WebView") || str.contains("WebViewClient") -> frameworks.add("Android WebView")
            str.startsWith("Lcom/unity3d/") -> frameworks.add("Unity")
            str.startsWith("Landroidx/appcompat/") || str.startsWith("Lcom/google/android/material/") -> frameworks.add("Android XML Views / Material")
        }
    }

    private fun descriptorToClassName(desc: String) = if (desc.startsWith("L") && desc.endsWith(";")) desc.substring(1, desc.length - 1).replace('/', '.') else ""
}

import androidx.lifecycle.viewModelScope
import com.example.engine.ApkAnalyzerEngine
import com.example.engine.ExportManager
import com.example.i18n.AppLanguage
import com.example.i18n.stringsFor
import com.example.model.*
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

enum class InspectionTab { TECH_STACK, FILES_CODE, HTML_REPORT, MANIFEST_DEX }

sealed interface ScreenState {
    data object Home : ScreenState
    data class Analyzing(val targetName: String) : ScreenState
    data class Result(val analysis: ApkAnalysisResult) : ScreenState
    data class Error(val message: String) : ScreenState
}

class MainViewModel(application: Application) : AndroidViewModel(application) {
    private val analyzer = ApkAnalyzerEngine(application)
    private val exporter = ExportManager(application)
    private val _language = MutableStateFlow(AppLanguage.KURDISH)
    val language: StateFlow<AppLanguage> = _language.asStateFlow()
    private val _isDarkTheme = MutableStateFlow(true)
    val isDarkTheme: StateFlow<Boolean> = _isDarkTheme.asStateFlow()
    private val _screenState = MutableStateFlow<ScreenState>(ScreenState.Home)
    val screenState: StateFlow<ScreenState> = _screenState.asStateFlow()
    private val _installedApps = MutableStateFlow<List<InstalledAppItem>>(emptyList())
    private val _appSearchQuery = MutableStateFlow("")
    val filteredApps = combine(_installedApps, _appSearchQuery) { apps, q ->
        if (q.isBlank()) apps else apps.filter { it.appName.contains(q, true) || it.packageName.contains(q, true) }
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    init { viewModelScope.launch { _installedApps.value = runCatching { analyzer.loadInstalledLauncherApps() }.getOrDefault(emptyList()) } }
    fun toggleLanguage() { _language.value = if (_language.value == AppLanguage.KURDISH) AppLanguage.ENGLISH else AppLanguage.KURDISH }
    fun toggleTheme() { _isDarkTheme.value = !_isDarkTheme.value }
    fun inspectFromUri(uri: Uri) {
        viewModelScope.launch {
            _screenState.value = ScreenState.Analyzing("APK")
            runCatching { analyzer.analyzeApk(analyzer.copyUriToInspectableApk(uri).absolutePath) }
                .onSuccess { _screenState.value = ScreenState.Result(it) }
                .onFailure { _screenState.value = ScreenState.Error(it.message ?: "Error") }
        }
    }
    fun downloadHtmlReport() {
        val current = (_screenState.value as? ScreenState.Result)?.analysis ?: return
        viewModelScope.launch { exporter.saveHtmlToDownloads(current) }
    }
}
